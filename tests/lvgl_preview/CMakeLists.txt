cmake_minimum_required(VERSION 3.16)
project(battery_screen_preview C)
get_filename_component(APP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT EXISTS "${APP_ROOT}/managed_components/lvgl__lvgl/CMakeLists.txt")
    message(FATAL_ERROR "Run the IDF dependency resolver first; LVGL 9.5.0 is required")
endif()
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${APP_ROOT}/managed_components/lvgl__lvgl" lvgl)
add_executable(battery_screen_preview ../battery_screen_preview.c
    "${APP_ROOT}/main/battery_ui.c" "${APP_ROOT}/main/battery_model.c" "${APP_ROOT}/main/battery_care.c")
target_include_directories(battery_screen_preview PRIVATE "${APP_ROOT}/main" "${APP_ROOT}/tests/bsp_stubs" "${APP_ROOT}/components/bsp/include")
target_compile_definitions(battery_screen_preview PRIVATE _POSIX_C_SOURCE=200809L)
target_link_libraries(battery_screen_preview lvgl m)
